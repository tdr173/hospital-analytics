%% Figure 4.19b + 4.24 — PrefixSpan end-to-end pipeline
%% Sequence shaping → mining → ranking → dedup → attribution.

flowchart TB
  classDef src fill:#bfdbfe,stroke:#1e3a8a,color:#000
  classDef shape fill:#fde68a,stroke:#a16207,color:#000
  classDef mine fill:#fecaca,stroke:#7f1d1d,color:#000
  classDef rank fill:#bbf7d0,stroke:#14532d,color:#000
  classDef out fill:#e9d5ff,stroke:#581c87,color:#000

  src["UNION ALL events from<br/>condition / medication / procedure / careplan<br/>with prefixes C: / M: / P: / CP:"]:::src

  src --> b1["Item blacklist<br/>(assessment of health and social care needs,<br/>medication review due,<br/>medication reconciliation)"]:::shape
  b1 --> b2["Sort by (patient_id, event_date, prefix_priority, item)<br/>prefix_priority: C&lt;M&lt;P&lt;CP"]:::shape
  b2 --> b3["2-year rolling window<br/>WINDOW_DAYS = 730<br/>(per patient, from latest event)"]:::shape
  b3 --> b4["Run-length collapse<br/>(prev_pid==pid AND prev_item==item) drop"]:::shape
  b4 --> b5["Tail-truncate to last 25 events per patient"]:::shape

  b5 --> e1["Integer-encode items<br/>(faster int compares in PrefixSpan)"]:::mine
  e1 --> e2["Compute single-item supports (used in GML)"]:::mine
  e1 --> m1["PrefixSpan.topk(K * 20, closed=True)<br/>minlen=3, maxlen=5<br/>(over-mining pool of 10,000)"]:::mine

  m1 --> f1["Drop patterns with sup &lt; 1% × N"]:::mine
  f1 --> f2["Reject 3+ consecutive<br/>same-type items"]:::mine

  f2 --> r1["GML scoring<br/>log-space:<br/>log_sup - log_N - mean_log_single_sup<br/>× (1 + 0.3·(t-1)) type-diversity bonus"]:::rank
  r1 --> r2["Closedness post-filter:<br/>drop P if longer Q with same support<br/>and P ⊆ Q (set + order)"]:::rank
  r2 --> r3["Greedy Jaccard dedup<br/>thresh=0.80<br/>prefix_cap=6<br/>suffix_cap=20<br/>dental_cap=15"]:::rank
  r3 --> r4["Truncate to TOP_K = 500"]:::rank

  r4 --> a1["Attribute patients<br/>1) build posting lists<br/>2) intersect by rarity (rarest first)<br/>3) two-pointer subsequence verify"]:::out
  a1 --> p1["Persist:<br/>prefix_span_pattern (rank, str, support)<br/>prefix_span_patient_pattern (M:N)"]:::out
