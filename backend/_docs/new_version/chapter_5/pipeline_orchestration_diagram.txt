  --
  CLUSTERING  (K-Prototypes + AI cluster cards)
  --
  SEQUENCES_GLOBAL  (PrefixSpan, full population)
  --
  SEQUENCES_CLUSTERS  (PrefixSpan per cluster 0..k-1, outliers -1)
  --
  DONE
}

rectangle "SUCCESS\n(last_run_at updated)" as SUCCESS
rectangle "FAILED\n(error message recorded)" as FAILED

GUARD   --> RUNNING : wins race\n(status -> RUNNING)
RUNNING --> SUCCESS : all stages complete
RUNNING --> FAILED  : any stage raises

SUCCESS --> IDLE : next trigger
FAILED  --> IDLE : next trigger
