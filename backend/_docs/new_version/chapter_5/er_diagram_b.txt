  BackgroundColor<<junction>> #DCEFD4
  BorderColor<<junction>> #4C7A3E
  BackgroundColor<<singleton>> #F6D9E4

  * patient_id : VARCHAR <<PK>>
  --
  full definition in Group A
  (see Figure 5.2)
}

entity cluster_meta <<singleton>> {
  * id : INT <<PK>>
  --
  SINGLETON -- exactly 1 row
  (TRUNCATE + INSERT on every re-run)
}

entity cluster <<analytics>> {
  * id : INT <<PK>>
  --
  one row per cluster label
  (0..k-1, and -1 = outliers)
}

entity cluster_patient <<analytics>> {
  * id : INT <<PK>>
  --
  # patient_id : VARCHAR <<FK>>
}

entity prefix_span_pattern <<analytics>> {
  * id : INT <<PK>>
  --
  cluster_label NULL = global scope,
  else per-cluster scope
}

entity prefix_span_patient_pattern <<junction>> {
  * pattern_id : INT <<PK, FK>>
  * patient_id : VARCHAR <<PK>>
}

entity pipeline_run <<singleton>> {
  * id : INT <<PK>>
  --
  SINGLETON -- exactly 1 row
  (orchestration state machine)
}

' -- hidden ordering links: pin the two "hub" pairs onto aligned
